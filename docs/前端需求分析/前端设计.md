### 3.2 路由守卫

```typescript
// router/guards.ts

import type { Router } from 'vue-router'
import { useAuthStore } from '@/stores/auth'

export function setupGuards(router: Router) {
  router.beforeEach(async (to, _from, next) => {
    const authStore = useAuthStore()

    // 1. 无需认证 → 直接放行
    if (!to.meta.requiresAuth) {
      // 已登录用户访问登录页 → 重定向到对应工作台
      if (to.name === 'Login' && authStore.isLoggedIn) {
        return next(getRoleDashboard(authStore.roles))
      }
      return next()
    }

    // 2. 需要认证但未登录 → 跳转登录页
    if (!authStore.isLoggedIn) {
      return next({ name: 'Login', query: { redirect: to.fullPath } })
    }

    // 3. 已登录 → 检查角色权限
    const allowedRoles = to.meta.roles as string[] | undefined
    if (allowedRoles && allowedRoles.length > 0) {
      const hasPermission = authStore.roles.some(r => allowedRoles.includes(r))
      if (!hasPermission) {
        // 角色不匹配 → 重定向到该用户的默认工作台
        return next(getRoleDashboard(authStore.roles))
      }
    }

    next()
  })
}

function getRoleDashboard(roles: string[]): string {
  if (roles.includes('TEACHER')) return '/teacher/dashboard'
  if (roles.includes('COLLEGE_REVIEWER')) return '/college/dashboard'
  if (roles.includes('OFFICE')) return '/office/dashboard'
  if (roles.includes('DEAN')) return '/dean/dashboard'
  return '/login'
}
```

```typescript
// api/index.ts

const http = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,   // /api
  timeout: 30000,
})

// 请求拦截器
http.interceptors.request.use(config => {
  const token = useAuthStore().token
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器
http.interceptors.response.use(
  response => response.data,                    // 统一解包 data
  error => {
    if (error.response?.status === 401) {
      // Token 过期 → 尝试刷新 → 失败则跳转登录
      return refreshTokenAndRetry(error)
    }
    if (error.response?.status === 403) {
      ElMessage.error('无操作权限')
    }
    return Promise.reject(error)
  }
)
```

### 8.1 FileUpload 文件上传组件

```
Props:
  - maxSize: number         (默认 20，单位 MB)
  - maxCount: number        (默认 10)
  - accept: string          (默认 ".doc,.docx,.pdf,.jpg,.png")
  - modelValue: FileInfo[]  (v-model 绑定)

Events:
  - update:modelValue
  - upload-success(file)
  - upload-error(error)
  - upload-progress(percent)

功能:
  - 拖拽上传 + 点击上传
  - 文件列表展示（名称、大小、上传进度、状态图标）
  - 断点续传（分片上传，> 10MB 自动分片）

  - 文件格式校验 + 大小校验
```

## 9. 关键交互流程

### 9.1 授课计划提交 → 审核流程

```
教师填写表单 → 上传附件 → 点击「提交审核」
  → 前端校验表单 + 文件
  → POST /api/teaching-plans
  → 成功后跳转列表页，状态显示「待学院审核」
  → 教师端可看到时间线第一条

学院审核员登录 → 审核列表看到该记录
  → 点击进入详情
  → 查看附件（在线预览/下载）
  → 填写审核意见（可选）
  → 点击「通过」：状态 →「待教务处审核」
  → 点击「驳回」：必须填写驳回理由 → 状态 →「待修改」
  → 教师收到通知

教务处登录 → 终审列表看到该记录
  → 点击进入详情
  → 查看原始材料 + 学院审核意见
  → 点击「通过」：状态 →「已通过」，归档
  → 点击「驳回」：选择驳回目标（教师/学院）+ 填写理由
  → 教师收到通知
```

### 9.2 AI 材料评审流程

```
教师选择材料类型 → 上传文件 → 选择关联课程标准 → 点击「开始AI评审」
  → POST /api/ai-evaluations
  → 后端异步调用火山引擎 LLM
  → 前端轮询 GET /api/ai-evaluations/{id}（每 3 秒）
  → 或通过 WebSocket 推送结果
  → 状态变为「AI评审完成」
  → 教师点击进入详情查看评分和建议

教务处 AI 评审复核列表 → 点击进入复核详情
  → 查看原始材料 + AI 评分详情
  → 判断评分是否合理
  → 「确认通过」：AI 评分生效，材料状态更新
  → 「驳回修改」：填写修改要求 + 截止日期，教师收到通知
  → 支持手动调整评分并注明理由
```

### 9.3 目录对齐分析流程

```
院长选择专业 + 人培方案版本 → 点击「开始分析」
  → POST /api/alignment/analyze { majorId, tcpId }
  → 后端异步调用火山引擎 LLM
  → 生成分析报告存入 alignment_reports 表
  → 前端展示报告
  → 院长可导出 PDF
  → 教师可在自己的工作台查看本院专业的分析报告（只读）
```

## 10. 权限控制方案

### 10.1 路由级权限

通过路由 `meta.roles` + 路由守卫 `guards.ts` 实现，拦截无权限访问。

### 10.2 按钮级权限

```html
<!-- v-permission 自定义指令 -->
<el-button v-permission="['OFFICE']" @click="approve">通过</el-button>
<el-button v-permission="['TEACHER']" @click="submit">提交</el-button>

<!-- 或通过 composable -->
<script setup>
const { hasRole } = usePermission()
</script>
<template>
  <el-button v-if="hasRole('OFFICE')" @click="approve">通过</el-button>
</template>
```

### 10.3 数据级权限

- 教师：仅查看自己的授课计划、评审记录
- 学院审核员：仅查看本学院的教师数据
- 院长：仅管理本学院的人培方案和课程标准
- 教务处：查看全校数据
- 前端通过 API 传参 + 后端数据过滤共同保证
